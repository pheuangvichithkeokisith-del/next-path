```python
# Multi-select normalization: capped at 3.0 pts to prevent broad answer inflation
raw_pts_q[C] = sum(POSITIVE_SIGNALS[q][opt].get(C, 0) for opt in valid_opts)
norm_q[C] = min(1.0, float(raw_pts_q[C]) / 3.0)
```
